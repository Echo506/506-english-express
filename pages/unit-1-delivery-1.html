"use strict"; document.addEventListener("DOMContentLoaded", () => { const storageKey = "506EnglishExpress.unit1.delivery1.progress"; const totalActivities = 9; const activityCards = Array.from( document.querySelectorAll(".activity-card[data-activity]") ); const progressText = document.getElementById("progress-text"); const progressFill = document.getElementById("progress-fill"); const progressTrack = document.querySelector(".progress-track"); const resetButton = document.getElementById("reset-progress"); const completeMessage = document.getElementById( "lesson-complete-message" ); function getCompletedActivities() { try { const saved = JSON.parse(localStorage.getItem(storageKey)); if (!Array.isArray(saved)) { return []; } return [...new Set( saved .map(Number) .filter( (activity) => Number.isInteger(activity) && activity >= 1 && activity <= totalActivities ) )].sort((a, b) => a - b); } catch { return []; } } function saveCompletedActivities(completedActivities) { localStorage.setItem( storageKey, JSON.stringify(completedActivities) ); } function getCard(activityNumber) { return document.querySelector( `.activity-card[data-activity="${activityNumber}"]` ); } function getFeedback(card) { return card.querySelector(".activity-feedback"); } function updateProgress() { const completedActivities = getCompletedActivities(); const completedCount = completedActivities.length; const percentage = Math.round( (completedCount / totalActivities) * 100 ); if (progressText) { progressText.textContent = `${completedCount} de ${totalActivities} secciones completadas`; } if (progressFill) { progressFill.style.width = `${percentage}%`; } if (progressTrack) { progressTrack.setAttribute("aria-valuenow", String(percentage)); } if (completeMessage) { completeMessage.hidden = completedCount !== totalActivities; } } function updateActivities() { const completedActivities = getCompletedActivities(); activityCards.forEach((card) => { const activityNumber = Number(card.dataset.activity); const previousActivityIsComplete = activityNumber === 1 || completedActivities.includes(activityNumber - 1); const activityIsComplete = completedActivities.includes(activityNumber); const feedback = getFeedback(card); const completeButton = card.querySelector(".complete-activity"); const finishButton = card.querySelector(".finish-lesson"); if (previousActivityIsComplete) { card.classList.remove("is-locked"); } else { card.classList.add("is-locked"); } if (activityIsComplete) { card.classList.add("is-complete"); if (feedback) { feedback.textContent = "✓ Actividad completada"; } if (completeButton) { completeButton.textContent = "Actividad completada"; } if (finishButton) { finishButton.textContent = "Entrega completada"; } } else { card.classList.remove("is-complete"); } }); updateProgress(); } function scrollToNextActivity(activityNumber) { const nextCard = getCard(activityNumber + 1); if (!nextCard) { if (completeMessage) { completeMessage.scrollIntoView({ behavior: "smooth", block: "center" }); } return; } setTimeout(() => { nextCard.scrollIntoView({ behavior: "smooth", block: "start" }); nextCard.classList.add("next-activity-highlight"); setTimeout(() => { nextCard.classList.remove("next-activity-highlight"); }, 1600); }, 250); } function completeActivity(activityNumber, shouldScroll = true) { const completedActivities = getCompletedActivities(); if (!completedActivities.includes(activityNumber)) { completedActivities.push(activityNumber); completedActivities.sort((a, b) => a - b); saveCompletedActivities(completedActivities); } updateActivities(); if (shouldScroll) { scrollToNextActivity(activityNumber); } } function showFeedback(card, message, isCorrect = false) { const feedback = getFeedback(card); if (!feedback) { return; } feedback.textContent = message; feedback.classList.toggle("feedback-correct", isCorrect); feedback.classList.toggle("feedback-error", !isCorrect); } function addNextButton(card) { const activityNumber = Number(card.dataset.activity); const actions = card.querySelector(".activity-actions"); if (!actions || activityNumber === totalActivities) { return; } if (actions.querySelector(".next-activity-button")) { return; } const nextButton = document.createElement("button"); nextButton.type = "button"; nextButton.className = "button button-outline next-activity-button"; nextButton.textContent = "Siguiente actividad →"; nextButton.addEventListener("click", () => { const completedActivities = getCompletedActivities(); if (!completedActivities.includes(activityNumber)) { completeActivity(activityNumber, false); } scrollToNextActivity(activityNumber); }); actions.appendChild(nextButton); } activityCards.forEach((card) => { addNextButton(card); }); document.querySelectorAll(".complete-activity").forEach((button) => { button.addEventListener("click", () => { const card = button.closest(".activity-card"); const activityNumber = Number(card.dataset.activity); completeActivity(activityNumber); }); }); document.querySelectorAll(".finish-lesson").forEach((button) => { button.addEventListener("click", () => { const card = button.closest(".activity-card"); const activityNumber = Number(card.dataset.activity); completeActivity(activityNumber); }); }); const comprehensionButton = document.querySelector( ".check-comprehension" ); if (comprehensionButton) { comprehensionButton.addEventListener("click", () => { const card = comprehensionButton.closest(".activity-card"); const questionOne = document.querySelector( 'input[name="question-1"]:checked' )?.value; const questionTwo = document.querySelector( 'input[name="question-2"]:checked' )?.value; if (questionOne === "b" && questionTwo === "b") { showFeedback( card, "✓ ¡Correcto! Jeff presenta a Tom y Tom es de California.", true ); completeActivity(2); } else { showFeedback( card, "Revisa el diálogo e inténtalo nuevamente." ); } }); } const vocabularyButton = document.querySelector(".check-vocabulary"); if (vocabularyButton) { vocabularyButton.addEventListener("click", () => { const card = vocabularyButton.closest(".activity-card"); const answer = document .getElementById("vocabulary-answer") .value .trim() .toLowerCase(); if ( answer.includes("tom") && answer.includes("new") && answer.includes("cameraman") ) { showFeedback( card, "✓ Correcto: Tom is the new cameraman.", true ); completeActivity(4); } else { showFeedback( card, "Pista: Tom is the new cameraman." ); } }); } const expressionsButton = document.querySelector(".check-expressions"); if (expressionsButton) { expressionsButton.addEventListener("click", () => { const card = expressionsButton.closest(".activity-card"); const greetingAnswer = document .getElementById("greeting-answer") .value .trim() .toLowerCase(); const originAnswer = document .getElementById("origin-answer") .value .trim() .toLowerCase(); const greetingCorrect = greetingAnswer === "hello"; const originCorrect = originAnswer === "am" || originAnswer === "'m"; if (greetingCorrect && originCorrect) { showFeedback( card, "✓ ¡Muy bien! Hello, nice to meet you. / I am from Costa Rica.", true ); completeActivity(5); } else { showFeedback( card, "Pista: Hello, nice to meet you. / I am from Costa Rica." ); } }); } if (resetButton) { resetButton.addEventListener("click", () => { const shouldReset = window.confirm( "¿Deseas reiniciar el progreso de esta entrega?" ); if (!shouldReset) { return; } localStorage.removeItem(storageKey); document .querySelectorAll("input[type='text'], textarea") .forEach((field) => { field.value = ""; }); document .querySelectorAll("input[type='radio']") .forEach((field) => { field.checked = false; }); document .querySelectorAll(".activity-feedback") .forEach((feedback) => { feedback.textContent = ""; }); document .querySelectorAll(".complete-activity") .forEach((button) => { button.textContent = button.closest('[data-activity="3"]') ? "Marcar práctica como completada" : button.closest('[data-activity="6"]') ? "Marcar recordatorio como completado" : button.closest('[data-activity="7"]') ? "Marcar práctica oral como completada" : button.closest('[data-activity="8"]') ? "Marcar lectura como completada" : "Marcar diálogo como completado"; }); const finishButton = document.querySelector(".finish-lesson"); if (finishButton) { finishButton.textContent = "Completar entrega"; } updateActivities(); const firstCard = getCard(1); if (firstCard) { firstCard.scrollIntoView({ behavior: "smooth", block: "start" }); } }); } updateActivities(); });
